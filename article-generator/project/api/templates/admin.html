Lightweight Reader Tracking Script
Embed this script in WordPress footer or article template to track views, clicks, and active reading time.

This tracking script uses browser visibility awareness and debounced heartbeats to calculate real reading duration (ignoring inactive tabs). 

<script>
(function() {
  var articleId = window.INPARTNER_ARTICLE_ID || null;
  var wpPostId = window.INPARTNER_WP_POST_ID || null;
  var trackUrl = "{{ request.base_url }}api/analytics/track";
  var readingSeconds = 0;
  var sessionId = localStorage.getItem("inp_sid") || (Math.random().toString(36).substring(2) + Date.now().toString(36));
  localStorage.setItem("inp_sid", sessionId);

  // 1. Record Pageview
  fetch(trackUrl, {
    method: "POST",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify({ article_id: articleId, wordpress_post_id: wpPostId, event_type: "view", session_id: sessionId, referrer: document.referrer })
  }).catch(function(){});

  // 2. Active Tab Reading Timer
  setInterval(function() {
    if (!document.hidden) {
      readingSeconds += 10;
      fetch(trackUrl, {
        method: "POST",
        headers: {"Content-Type": "application/json"},
        body: JSON.stringify({ article_id: articleId, wordpress_post_id: wpPostId, event_type: "reading_heartbeat", reading_seconds: readingSeconds, session_id: sessionId })
      }).catch(function(){});
    }
  }, 10000);
})();
</script>
